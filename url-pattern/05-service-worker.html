URLPattern 在 Service Worker 中使用

说明： URLPattern 最初就是为 Service Worker 设计的，用于高效地匹配和路由请求。

1. 基本请求拦截

// sw.js - Service Worker 文件
const apiPattern = new URLPattern({ pathname: '/api/*' });
const staticPattern = new URLPattern({ pathname: '/static/*' });
const imagePattern = new URLPattern({ pathname: '*.{png,jpg,gif,webp}' });

self.addEventListener('fetch', (event) => {
  const url = event.request.url;

  if (apiPattern.test(url)) {
    // 处理 API 请求
    event.respondWith(handleApiRequest(event.request));
  } else if (staticPattern.test(url)) {
    // 缓存优先策略
    event.respondWith(cacheFirst(event.request));
  } else if (imagePattern.test(url)) {
    // 图片缓存策略
    event.respondWith(cacheWithFallback(event.request));
  }
});

2. API 版本路由

// 匹配不同版本的 API
const apiV1Pattern = new URLPattern({ pathname: '/api/v1/*' });
const apiV2Pattern = new URLPattern({ pathname: '/api/v2/*' });

self.addEventListener('fetch', (event) => {
  const url = event.request.url;

  if (apiV1Pattern.test(url)) {
    // V1 API - 可能需要转换响应格式
    event.respondWith(handleV1Api(event.request));
  } else if (apiV2Pattern.test(url)) {
    // V2 API - 直接转发
    event.respondWith(fetch(event.request));
  }
});

3. 动态路由参数提取

const userPattern = new URLPattern({ pathname: '/api/users/:userId' });
const postPattern = new URLPattern({ pathname: '/api/users/:userId/posts/:postId' });

self.addEventListener('fetch', (event) => {
  const url = event.request.url;

  // 检查用户详情请求
  const userMatch = userPattern.exec(url);
  if (userMatch) {
    const userId = userMatch.pathname.groups.userId;
    event.respondWith(getUserFromCache(userId));
    return;
  }

  // 检查文章详情请求
  const postMatch = postPattern.exec(url);
  if (postMatch) {
    const { userId, postId } = postMatch.pathname.groups;
    event.respondWith(getPostFromCache(userId, postId));
    return;
  }
});

4. 缓存策略路由

// 定义不同资源的缓存策略
const routes = [
  {
    pattern: new URLPattern({ pathname: '/api/config' }),
    strategy: 'network-first',  // 配置优先从网络获取
    cacheName: 'config-cache'
  },
  {
    pattern: new URLPattern({ pathname: '/api/users/*' }),
    strategy: 'stale-while-revalidate',  // 用户数据使用 SWR
    cacheName: 'users-cache'
  },
  {
    pattern: new URLPattern({ pathname: '/static/*' }),
    strategy: 'cache-first',  // 静态资源缓存优先
    cacheName: 'static-cache'
  },
  {
    pattern: new URLPattern({ pathname: '*.{js,css}' }),
    strategy: 'cache-first',
    cacheName: 'assets-cache'
  }
];

self.addEventListener('fetch', (event) => {
  const url = event.request.url;

  for (const route of routes) {
    if (route.pattern.test(url)) {
      event.respondWith(
        handleWithStrategy(event.request, route.strategy, route.cacheName)
      );
      return;
    }
  }
});

5. 离线页面回退

const htmlPattern = new URLPattern({ pathname: '*' });

self.addEventListener('fetch', (event) => {
  // 只处理 HTML 页面请求
  if (event.request.mode === 'navigate') {
    event.respondWith(
      fetch(event.request)
        .catch(() => {
          // 网络失败时返回离线页面
          return caches.match('/offline.html');
        })
    );
  }
});

6. 完整示例：Service Worker 路由器

// sw.js
class ServiceWorkerRouter {
  constructor() {
    this.routes = [];
  }

  get(pattern, handler) {
    this.routes.push({
      method: 'GET',
      pattern: new URLPattern(pattern),
      handler
    });
  }

  post(pattern, handler) {
    this.routes.push({
      method: 'POST',
      pattern: new URLPattern(pattern),
      handler
    });
  }

  handle(event) {
    const { request } = event;

    for (const route of this.routes) {
      if (route.method !== request.method) continue;

      const match = route.pattern.exec(request.url);
      if (match) {
        return route.handler(request, match);
      }
    }

    return fetch(request);
  }
}

// 使用路由器
const router = new ServiceWorkerRouter();

router.get({ pathname: '/api/users/:id' }, async (request, match) => {
  const userId = match.pathname.groups.id;
  const cache = await caches.open('users');
  const cached = await cache.match(request);

  if (cached) return cached;

  const response = await fetch(request);
  cache.put(request, response.clone());
  return response;
});

router.get({ pathname: '/static/*' }, async (request) => {
  const cache = await caches.open('static');
  return (await cache.match(request)) || fetch(request);
});

self.addEventListener('fetch', (event) => {
  event.respondWith(router.handle(event));
});
注意： 此页面展示的是 Service Worker 中的代码示例。 要实际运行这些代码，需要创建单独的 sw.js 文件并在主页面中注册。

注册 Service Worker

// 在主页面中注册 Service Worker
if ('serviceWorker' in navigator) {
  navigator.serviceWorker.register('/sw.js')
    .then(registration => {
      console.log('SW registered:', registration);
    })
    .catch(error => {
      console.log('SW registration failed:', error);
    });
}
