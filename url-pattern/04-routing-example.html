URLPattern 路由实战示例

简单路由器实现

class Router {
  constructor(baseURL) {
    this.baseURL = baseURL;
    this.routes = [];
  }

  add(pattern, handler) {
    this.routes.push({
      pattern: new URLPattern(pattern, this.baseURL),
      handler
    });
  }

  match(url) {
    for (const route of this.routes) {
      const result = route.pattern.exec(url);
      if (result) {
        return { handler: route.handler, params: result.pathname.groups };
      }
    }
    return null;
  }
}

// 当前演示使用的路由配置
router.add('/', () => '首页');
router.add('/users', () => '用户列表页');
router.add('/users/:id', (params) => `用户详情: ${params.id}`);
router.add('/users/:userId/posts', (params) => `用户 ${params.userId} 的文章列表`);
router.add('/users/:userId/posts/:postId', (params) =>
  `用户 ${params.userId} 的文章 ${params.postId}`);
router.add('/products', () => '产品列表页');
router.add('/products/:id(\\d+)', (params) => `产品详情: ${params.id}`);
router.add('/about', () => '关于页面');
router.add('/*', () => '404 - 页面未找到');

API 路由示例

// API 路由器
const apiRouter = new Router('https://api.example.com');

apiRouter.add('/v:version/users', (params) => {
  return { version: params.version, resource: 'users' };
});

apiRouter.add('/v:version/users/:id', (params) => {
  return { version: params.version, userId: params.id };
});

apiRouter.add('/v:version/users/:userId/posts/:postId', (params) => {
  return {
    version: params.version,
    userId: params.userId,
    postId: params.postId
  };
});

// 使用
const result = apiRouter.match('https://api.example.com/v2/users/123/posts/456');
// { version: '2', userId: '123', postId: '456' }
