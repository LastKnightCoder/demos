🎯 文件类型过滤

按类型选择文件

使用 types 选项限制可选择的文件类型。

代码示例

// 图片文件过滤
const imageOptions = {
  types: [{
    description: '图片文件',
    accept: {
      'image/*': ['.png', '.jpg', '.jpeg', '.gif', '.webp', '.svg']
    }
  }],
  excludeAcceptAllOption: true  // 隐藏"所有文件"选项
};

// 文本文件过滤
const textOptions = {
  types: [{
    description: '文本文件',
    accept: {
      'text/plain': ['.txt'],
      'text/markdown': ['.md']
    }
  }]
};

// 代码文件过滤（多种类型）
const codeOptions = {
  types: [
    { description: 'JavaScript', accept: { 'text/javascript': ['.js', '.mjs'] } },
    { description: 'TypeScript', accept: { 'text/typescript': ['.ts', '.tsx'] } },
    { description: 'HTML', accept: { 'text/html': ['.html', '.htm'] } },
    { description: 'CSS', accept: { 'text/css': ['.css', '.scss'] } }
  ]
};
