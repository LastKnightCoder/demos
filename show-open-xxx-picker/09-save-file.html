💾 文件保存演示

使用 showSaveFilePicker() 保存文件到本地

1. 保存文本内容

输入文本内容，然后保存为本地文件。

2. 保存 JSON 数据

编辑 JSON 数据并保存。

3. 保存 Canvas 绘图

在画布上绘制内容，然后保存为图片。

4. 多格式导出

选择不同的格式导出相同的数据。

代码示例

// 基本保存文件
async function saveFile(content, suggestedName, types) {
  const handle = await window.showSaveFilePicker({
    suggestedName,
    types
  });
  const writable = await handle.createWritable();
  await writable.write(content);
  await writable.close();
  return handle.name;
}

// 保存 Canvas 为图片
async function saveCanvasAsImage(canvas, format = 'png') {
  const mimeType = format === 'jpg' ? 'image/jpeg' : 'image/png';
  const blob = await new Promise(r => canvas.toBlob(r, mimeType, 0.9));

  const handle = await window.showSaveFilePicker({
    suggestedName: `drawing.${format}`,
    types: [{
      description: '图片文件',
      accept: { [mimeType]: [`.${format}`] }
    }]
  });

  const writable = await handle.createWritable();
  await writable.write(blob);
  await writable.close();
}
